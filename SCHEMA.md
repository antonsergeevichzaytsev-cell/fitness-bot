  "sets": [

      "id": "s_<sha1 hash>",           # уникальный ID сета
      "date": "2026-07-28",             # дата тренировки (не подхода — сеты одной тренировки делят дату)
      "exercise": "жим лёжа гантели",   # нормализованное название (нижний регистр, без лишних слов)
      "exercise_raw": "жим лежа гантелями",  # как написал Антон — для отладки парсинга
      "weight_kg": 30.0,                # вес одной единицы (гантели/штанги), null если без веса
      "reps": 10,                       # повторов в этом подходе
      "set_number": 1,                  # номер подхода в рамках упражнения в этой тренировке (1, 2, 3...)
      "rpe": null,                      # Rate of Perceived Exertion 1-10, если Антон указал ("тяжело" -> 8-9, "легко" -> 5-6) — опционально
      "note": "",                       # свободный текст, если Антон добавил комментарий
      "safety_status": "ok",            # результат safety.check_exercise на момент записи — "ok"/"hard_block"/"manual_progression_only"
      "ts": "2026-07-28T18:32:00+00:00" # когда сообщение пришло в Telegram (для сортировки внутри дня)
    }
  ],

  "exercise_aliases": {
    # DeepSeek парсит свободный текст, разные формулировки одного упражнения
    # должны схлопываться в одно нормализованное имя, иначе прогрессия

    # автоматически при парсинге: новая формулировка -> сверяется с уже
    # известными через DeepSeek (embedding-подобная проверка через LLM,
    # не точное строковое сравнение) -> если похоже на существующее,
    # используется существующее нормализованное имя.
    "жим лёжа гантели": ["жим лежа гантелями", "гантели лежа", "dumbbell bench"],
    "присед со штангой": ["присед штанга", "back squat"]
  },

  "pending_suggestions": [
    # Предложения прогрессии, ждущие подтверждения (👍/👎 в Telegram).
    # Тот же паттерн, что msg_map в filings.py — message_id связывает
    # кнопку с конкретным предложением.
    {
      "id": "sugg_<hash>",
      "exercise": "жим лёжа гантели",
      "current_weight_kg": 30.0,
      "current_reps": 10,
      "suggested_weight_kg": 32.5,
      "suggested_reps": 10,
      "reasoning": "3 тренировки подряд 30кг x10 без затруднений — пора поднять вес",
      "message_id": null,               # заполняется после отправки в Telegram
      "status": "pending",              # pending / confirmed / rejected
      "created_ts": "2026-07-28T18:35:00+00:00"
    }
  ],

  "targets": {
    # Целевые вес/повторы по нормализованному имени упражнения — то,
    # что бот будет предлагать делать НА СЛЕДУЮЩЕЙ тренировке. Меняется
    # только при confirmed предложении (пользователь нажал 👍), не
    # автоматически при каждом парсинге. Если target для упражнения ещё

    "жим лёжа гантели": {"weight_kg": 30.0, "reps": 10, "set_at": "2026-07-28"}
  }

=== Принципы дизайна ===

1. Один сет = одна запись, не агрегат. "жим лёжа гантели 30х10х3" (3 подхода)
   парсится в 3 отдельные записи с set_number 1/2/3 — так проще считать
   тренды по последнему подходу отдельно от первого (обычно последний
   подход тяжелее переносится, это сигнал для прогрессии).

2. exercise (нормализованное) отделено от exercise_raw (как написал Антон).
   Без этого история упражнения расползается на "жим лежа гантелями",
   "гантели лежа", "жим гантелями лёжа" — три разных "упражнения" в
   статистике вместо одного с полной историей.

3. safety_status сохраняется на момент записи, не пересчитывается заново
   каждый раз — если правила safety_constraints.json позже изменятся,
   историческая запись не должна тихо "перекраситься" в другой статус
   задним числом, это исказило бы историю решений.

4. pending_suggestions — предложения прогрессии живут отдельно от sets,
   не смешиваются с фактической историей выполнения. Подтверждённое
   предложение НЕ создаёт новую запись в sets само по себе — запись в
   sets появляется только когда Антон реально сделал подход и написал
   об этом. Предложение лишь меняет "целевые" вес/повторы для следующей
   тренировки (хранится отдельно, target_weight/target_reps по exercise).
